Arguments (pass arguments as '<name>:=<value>'):

    'rviz_fixed_frame':
        Use map from startup for SLAM/AMCL retained-map visualization. Valid choices are: ['odom', 'map']
        (default: 'odom')

    'world':
        One of: ['depot', 'maze', 'warehouse']
        (default: 'warehouse')

    'model':
        One of: ['standard', 'lite']
        (default: 'standard')

    'namespace':
        Robot ROS namespace
        (default: '')

    'rviz':
        One of: ['true', 'false']
        (default: 'false')
