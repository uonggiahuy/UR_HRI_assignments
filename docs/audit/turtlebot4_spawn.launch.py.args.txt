Arguments (pass arguments as '<name>:=<value>'):

    'rviz':
        Start rviz. Valid choices are: ['true', 'false']
        (default: 'false')

    'use_sim_time':
        use_sim_time. Valid choices are: ['true', 'false']
        (default: 'true')

    'model':
        Turtlebot4 Model. Valid choices are: ['standard', 'lite']
        (default: 'standard')

    'namespace':
        Robot namespace
        (default: '')

    'localization':
        Whether to launch localization. Valid choices are: ['true', 'false']
        (default: 'false')

    'slam':
        Whether to launch SLAM. Valid choices are: ['true', 'false']
        (default: 'false')

    'nav2':
        Whether to launch Nav2. Valid choices are: ['true', 'false']
        (default: 'false')

    'x':
        x component of the robot pose.
        (default: '0.0')

    'y':
        y component of the robot pose.
        (default: '0.0')

    'z':
        z component of the robot pose.
        (default: '0.0')

    'yaw':
        yaw component of the robot pose.
        (default: '0.0')

    'param_file':
        Turtlebot4 Robot param file
        (default: PathJoinSubstitution(''/opt/ros/humble/share/turtlebot4_ignition_bringup', 'config', 'turtlebot4_node.yaml''))

    'robot_name':
        Robot name
        (default: 'turtlebot4')

    'gazebo':
        Which gazebo simulation to use. Valid choices are: ['classic', 'ignition']
        (default: 'classic')

    'visualize_rays':
        Enable/disable ray visualization. Valid choices are: ['true', 'false']
        (default: 'true')

    'dock_name':
        Ignition model name
        (default: 'standard_dock')

    'world':
        World name
        (default: 'warehouse')

    'params':
        Localization parameters
        (default: PathJoinSubstitution(''/opt/ros/humble/share/turtlebot4_navigation', 'config', 'localization.yaml''))

    'map':
        Full path to map yaml file to load
        (default: PathJoinSubstitution(''/opt/ros/humble/share/turtlebot4_navigation', 'maps', 'warehouse.yaml''))

    'params_file':
        Full path to the ROS2 parameters file to use for all launched nodes
        (default: '/opt/ros/humble/share/nav2_bringup/params/nav2_params.yaml')

    'autostart':
        Automatically startup the nav2 stack
        (default: 'true')

    'use_composition':
        Use composed bringup if True
        (default: 'False')

    'container_name':
        the name of conatiner that nodes will load in if use composition
        (default: 'nav2_container')

    'use_respawn':
        Whether to respawn if a node crashes. Applied when composition is disabled.
        (default: 'False')

    'log_level':
        log level
        (default: 'info')

    'sync':
        Use synchronous SLAM. Valid choices are: ['true', 'false']
        (default: 'true')

    'description':
        Launch turtlebot4 description
        (default: 'false')
